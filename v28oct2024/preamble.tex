% Define a new command for the centered title
\newcommand{\chaptertitle}[1]{%
  \newpage
  \vfill
  {\centering\Huge\bfseries #1}
  \vfill
  \thispagestyle{empty}
  \newpage
}

\titleformat{\section}
    {\centering\Huge\bfseries}
    {Chapter \thesection}
    {1em}
    {\chaptertitle}
